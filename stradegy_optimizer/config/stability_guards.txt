# Stability Guards Configuration (Spec 3.3.1)
# Implements freeze, rollback, and learning memory guards

stability_guards:
  # Freeze Logic (Spec 3.3.1)
  # "If configuration achieves profit > 5% and drawdown decreases for 3 consecutive iterations,
  # parameters must be frozen for N iterations (default: 5)"
  freeze_profit_threshold: 5.0          # Profit threshold % for freeze eligibility
  consecutive_improving_for_freeze: 3   # Number of consecutive improving iterations before freeze
  freeze_duration: 5                    # Number of iterations to keep parameters frozen
  
  # Rollback Logic (Spec 3.3.1)
  # "If drawdown increases by more than 15% in a single iteration,
  # the system must revert to the last known stable configuration"
  rollback_drawdown_spike: 15.0         # Drawdown increase % that triggers rollback
  
  # Fallback Logic (Spec 3.3.1)
  # "If the AI API fails, the system must apply deterministic heuristic adjustments"
  fallback_heuristic_enabled: true
  
  # Learning Memory & Blacklist (Spec 3.3.1)
  # "Blacklisted parameter sets that caused drawdown > 40%"
  blacklist_drawdown_threshold: 40.0    # Drawdown % that triggers blacklisting
  
  # Memory tracking
  max_memory_size: 1000                 # Maximum parameter sets to remember
  memory_persistence_enabled: true      # Save memory to disk
